const BACKEND_URL = import.meta.env.VITE_BACKEND_URL || ""; // Défini dans .env.local ou variable Vercel

const T = {
  bg:"#F4F6F9", surface:"#FFFFFF", surfaceAlt:"#EEF1F6",
  border:"#DDE3EE", borderStrong:"#C5CFDF",
  cyan:"#0099CC", cyanLight:"#E0F5FF", cyanMid:"#B3E5F5",
  text:"#1A2233", textMid:"#4A5568", textSoft:"#8A96A8",
  green:"#0B9E5E", greenBg:"#E6F7F0", greenBorder:"#A8DFC5",

  purple:"#6B3FA0", purpleBg:"#F3EEFF", purpleBorder:"#C9B0E8",

async function fetchPlate(plate) {
  const url = `${BACKEND_URL}/api/plaque?immat=${encodeURIComponent(plate)}`;
  const resp = await fetch(url);
  if (!resp.ok) throw new Error(`Erreur serveur HTTP ${resp.status}`);
  return resp.json();
}

function normalize(raw) {
  const d = raw?.data || raw || {};
  if (d.erreur && d.erreur !== "") return null;
  if (!d.marque && !d.modele) return null;
  const energieMap = {"1":"Essence","2":"Diesel","3":"Électrique","4":"Hybride","5":"E85","6":"Bicarburation"};
  return {
    immat:        d.immat || "—",
    marque:       d.marque || "—",
    modele:       d.modele || "—",
    version:      d.sra_commercial || "",
    date1erCir:   d.date1erCir_fr || "—",
    vin:          d.vin || "—",
    couleur:      d.couleur || "—",
    energie:      d.energieNGC || energieMap[String(d.energie)] || "—",
    genreVCG:     d.genreVCGNGC || "—",
    carrosserie:  d.carrosserieCG || "—",
    puisFisc:     d.puisFisc ? `${d.puisFisc} CV` : "—",
    puisKW:       d.puisFiscReelKW || d.puisFiscReel || "",
    puisCH:       d.puisFiscReelCH || "",
    co2:          d.co2 ? `${d.co2} g/km` : "—",
    cylindres:    String(d.cylindres || "—"),
    ccm:          d.ccm || "",
    boite:        d.boite_vitesse==="M"?"Manuelle":d.boite_vitesse==="A"?"Automatique":(d.boite_vitesse||"—"),
    nbPassagers:  String(d.nr_passagers || "—"),
    nbPortes:     String(d.nb_portes || "—"),
    poids:        d.poids || "—",
    ptac:         d.ptac ? `${d.ptac} kg` : "—",
    typeMine:     d.type_mine || d.cnit || "—",
    codeMoteur:   d.code_moteur || "—",
    sraId:        d.sra_id || "—",
    sraGroup:     d.sra_group || "",
    ktype:        d.k_type || d.tecdoc_carid || "—",
    numSerie:     d.numero_serie || "—",
    numReception: d.num_reception || "—",
    variante:     d.variante || "—",
    logoUrl:      d.logo_marque || null,
    collection:   d.collection === "oui",
  };
}

function ScanBar({active}) {
  return (
    <div style={{position:"absolute",inset:0,overflow:"hidden",borderRadius:"10px",pointerEvents:"none",opacity:active?1:0,transition:"opacity 0.3s"}}>
      <div style={{position:"absolute",left:0,right:0,height:"3px",background:`linear-gradient(90deg,transparent,${T.cyan},${T.cyan},transparent)`,boxShadow:`0 0 16px ${T.cyan}88`,animation:active?"scan 1.4s ease-in-out infinite":"none"}}/>
      <style>{`@keyframes scan{0%{top:0%}50%{top:95%}100%{top:0%}} @keyframes pulse{0%,100%{opacity:1}50%{opacity:0.4}}`}</style>
    </div>
  );
}

function Badge({label,color}) {
  const m={green:{bg:T.greenBg,text:T.green,border:T.greenBorder},red:{bg:T.redBg,text:T.red,border:T.redBorder},orange:{bg:T.orangeBg,text:T.orange,border:T.orangeBorder},cyan:{bg:T.cyanLight,text:T.cyan,border:T.cyanMid},gray:{bg:T.surfaceAlt,text:T.textSoft,border:T.border},purple:{bg:T.purpleBg,text:T.purple,border:T.purpleBorder}};
  const c=m[color]||m.gray;
  return <span style={{background:c.bg,color:c.text,border:`1px solid ${c.border}`,borderRadius:"5px",padding:"2px 8px",fontSize:"11px",fontFamily:"monospace",fontWeight:700,letterSpacing:"0.05em",textTransform:"uppercase"}}>{label}</span>;
}

function Section({title,icon,children}) {
  return (
    <div style={{background:T.surface,borderRadius:"12px",border:`1px solid ${T.border}`,overflow:"hidden",marginBottom:"14px",boxShadow:"0 1px 4px rgba(0,0,0,0.06)"}}>
      <div style={{padding:"13px 18px",borderBottom:`1px solid ${T.border}`,display:"flex",alignItems:"center",gap:"10px",background:T.surfaceAlt}}>
        <span style={{fontSize:"16px"}}>{icon}</span>
        <span style={{fontWeight:700,fontSize:"12px",letterSpacing:"0.12em",textTransform:"uppercase",color:T.cyan}}>{title}</span>
        <div style={{flex:1,height:"1px",background:T.border,marginLeft:"6px"}}/>
      </div>
      <div style={{padding:"16px 18px"}}>{children}</div>
    </div>
  );
}

function Row({label,value,accent}) {
  if(!value||value==="—"||value==="")return null;
  return (
    <div style={{display:"flex",justifyContent:"space-between",alignItems:"flex-start",padding:"9px 0",borderBottom:`1px solid ${T.border}`,gap:"16px"}}>
      <span style={{fontSize:"12px",color:T.textSoft,flexShrink:0,maxWidth:"50%"}}>{label}</span>
      <span style={{fontSize:"13px",fontWeight:700,color:accent?T.cyan:T.text,fontFamily:"monospace",textAlign:"right",wordBreak:"break-all"}}>{value}</span>
    </div>
  );
}

function ExtLink({href,icon,label,sub,color="cyan"}) {
  const m={cyan:{border:T.cyanMid,text:T.cyan,bg:T.cyanLight},green:{border:T.greenBorder,text:T.green,bg:T.greenBg},orange:{border:T.orangeBorder,text:T.orange,bg:T.orangeBg},purple:{border:T.purpleBorder,text:T.purple,bg:T.purpleBg}};
  const c=m[color]||m.cyan;
  return (
    <a href={href} target="_blank" rel="noopener noreferrer" style={{textDecoration:"none",display:"block",marginBottom:"8px"}}>
      <div style={{display:"flex",alignItems:"center",gap:"12px",padding:"11px 14px",background:T.surface,border:`1px solid ${T.border}`,borderRadius:"10px",transition:"all 0.2s",cursor:"pointer"}}
        onMouseEnter={e=>{e.currentTarget.style.borderColor=c.border;e.currentTarget.style.background=c.bg;}}
        onMouseLeave={e=>{e.currentTarget.style.borderColor=T.border;e.currentTarget.style.background=T.surface;}}>
        <span style={{fontSize:"18px"}}>{icon}</span>
        <div style={{flex:1}}>
          <div style={{fontWeight:700,fontSize:"13px",color:T.text}}>{label}</div>
          {sub&&<div style={{fontSize:"11px",color:T.textSoft,marginTop:"1px"}}>{sub}</div>}
        </div>
        <span style={{color:c.text,fontWeight:700}}>↗</span>
      </div>
    </a>
  );
}
